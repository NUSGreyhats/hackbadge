/*
 * Microphone ADC Reader for CH32V003
 * Reads analog signal from PA2 (ADC Channel 0) using DMA
 * Outputs readings via debug printf (minichlink -T)
 *
 * LED Volume Indicator:
 * - Orange (PC7): Baseline/Silence
 * - Yellow (PC6): Quiet sound
 * - Blue (PC4): Medium sound
 * - Purple (PC2): Loud sound
 *
 * Setup:
 * 1. Create funconfig.h with:
 *    #define FUNCONF_USE_DEBUGPRINTF 1
 * 2. Flash: make flash
 * 3. Monitor: minichlink -T
 */

#include "ch32fun.h"
#include <stdio.h>

 // LED pin definitions (active LOW - write LOW to turn on)
#define LED_PURPLE   PC2
#define LED_BLUE     PC4
#define LED_YELLOW   PC6
#define LED_ORANGE   PC7

#define LED_ON   FUN_LOW
#define LED_OFF  FUN_HIGH

#define ADC_NUMCHLS 1  // Only reading one channel (PA2)

volatile uint16_t adc_buffer[ADC_NUMCHLS];

/*
 * Initialize ADC with DMA for PA2 (Channel 0)
 */
void adc_init(void)
{
    // ADCCLK = 24 MHz => RCC_ADCPRE = 0: divide by 2
    RCC->CFGR0 &= ~(0x1F << 11); 

    // Enable GPIOA and ADC
    RCC->APB2PCENR |= RCC_APB2Periph_GPIOA | RCC_APB2Periph_ADC1; // APB2 Peripheral Clock Enable Register. |= is short for APB2PCENR= APB2PCENR | (Mask) 
    // Configures ADC Clock pre scaler. RCC = Reset and Clock Control register structure. CFGR0 = Configuration Register 0.
    // The field controlling the ADC clock division is at bits 11-14. 0x1F, 0b11111, << 11 creates a mask to clear the relevant 5 bits (bits 11-15) using the ~ operator. 
    // By clearing these bits, the ADC clock is typically set to divide the APB2 (Advanced Peripheral Bus 2) clock by 2, resulting in a suitable ADC clock frequency 24MHz
    // APB2 connects high-speed peripherals, such as the ADC, GPIO ports, and some timers, to the core of the microcontroller.
    // for ADC pre-scaler bits, 0b0000=0=/2, 0b0001=1=/4, 0b0010=2=/6...

    // PA2 is analog input channel 0
    GPIOA->CFGLR &= ~(0xf << (4 * 2));  // CNF = 00: Analog, MODE = 00: Input 
    // CFGLR (Configuration Register Low) controls pins 0-7. 4*2 is 8, which is the starting bit for Pin 2's configuration field (4 bits per pin). 
    // 0xf << 8 clears the 4 configuration bits for PA2. Setting them to 0b0000 for Analog Input Mode, configures PA2 to pass its analog voltage directly to the ADC.

    // Reset the ADC to init all regs, APB2 Peripheral Reset Register
    RCC->APB2PRSTR |= RCC_APB2Periph_ADC1; // Each bit corresponds to a peripheral on the APB2 bus. Setting a bit to '1' applies a hard-reset signal to that peripheral.
    RCC->APB2PRSTR &= ~RCC_APB2Periph_ADC1;

    // Set up single conversion on channel 0 (PA2 = A0)
    ADC1->RSQR1 = (ADC_NUMCHLS - 1) << 20;  // 1 channel in the sequence
    ADC1->RSQR2 = 0;
    ADC1->RSQR3 = (0 << (5 * 0));  // Channel 0 for PA2

    // Set sampling time for channel 0
    // 7 = 241 cycles (slowest, most stable) - important for RC-biased microphone
    ADC1->SAMPTR2 = (7 << (3 * 0));

    // Turn on ADC
    ADC1->CTLR2 |= ADC_ADON;

    // Reset calibration
    ADC1->CTLR2 |= ADC_RSTCAL;
    while (ADC1->CTLR2 & ADC_RSTCAL);

    // Calibrate
    ADC1->CTLR2 |= ADC_CAL;
    while (ADC1->CTLR2 & ADC_CAL);

    // Turn on DMA
    RCC->AHBPCENR |= RCC_AHBPeriph_DMA1;

    // DMA1_Channel1 is for ADC
    DMA1_Channel1->PADDR = (uint32_t)&ADC1->RDATAR;
    DMA1_Channel1->MADDR = (uint32_t)adc_buffer;
    DMA1_Channel1->CNTR = ADC_NUMCHLS;
    DMA1_Channel1->CFGR =
        DMA_M2M_Disable |
        DMA_Priority_VeryHigh |
        DMA_MemoryDataSize_HalfWord |
        DMA_PeripheralDataSize_HalfWord |
        DMA_MemoryInc_Enable |
        DMA_Mode_Circular |
        DMA_DIR_PeripheralSRC;

    // Turn on DMA channel 1
    DMA1_Channel1->CFGR |= DMA_CFGR1_EN;

    // Disable SCAN mode (only needed for multiple channels)
    ADC1->CTLR1 &= ~ADC_SCAN;

    // Clear EXTSEL and enable continuous conversion with DMA
    ADC1->CTLR2 &= ~(0b111 << 17);  // Clear external trigger select
    ADC1->CTLR2 |= ADC_CONT | ADC_DMA;  // Continuous mode + DMA

    // Start conversion
    ADC1->CTLR2 |= ADC_SWSTART;
}

/*
 * Update LED indicator based on audio level
 * LEDs are active LOW (write LOW to turn on)
 */
void update_leds(uint16_t deviation)
{
    // Turn off all LEDs first
    funDigitalWrite(LED_ORANGE, LED_OFF);
    funDigitalWrite(LED_YELLOW, LED_OFF);
    funDigitalWrite(LED_BLUE, LED_OFF);
    funDigitalWrite(LED_PURPLE, LED_OFF);

    // Light up LEDs based on volume level
    if (deviation < 20) {
        // Silence - only orange
        funDigitalWrite(LED_ORANGE, LED_ON);
    }
    else if (deviation < 50) {
        // Quiet - orange + yellow
        funDigitalWrite(LED_ORANGE, LED_ON);
        funDigitalWrite(LED_YELLOW, LED_ON);
    }
    else if (deviation < 100) {
        // Medium - orange + yellow + blue
        funDigitalWrite(LED_ORANGE, LED_ON);
        funDigitalWrite(LED_YELLOW, LED_ON);
        funDigitalWrite(LED_BLUE, LED_ON);
    }
    else {
        // Loud - all LEDs
        funDigitalWrite(LED_ORANGE, LED_ON);
        funDigitalWrite(LED_YELLOW, LED_ON);
        funDigitalWrite(LED_BLUE, LED_ON);
        funDigitalWrite(LED_PURPLE, LED_ON);
    }
}

int main(void)
{
    SystemInit();

    // Wait for debug printf to be ready
    while (!DebugPrintfBufferFree());

    Delay_Ms(100);

    printf("\r\n\r\nMicrophone ADC Reader\n");
    printf("Reading PA2 (ADC Channel 0 - A0)\n");
    printf("Debug output via minichlink -T\n\n");

    // Initialize ADC
    printf("Initializing ADC...");
    adc_init();
    printf("done.\n");

    funGpioInitAll();

    // Configure LEDs as outputs (active LOW)
    funPinMode(LED_PURPLE, GPIO_Speed_10MHz | GPIO_CNF_OUT_PP);
    funPinMode(LED_BLUE, GPIO_Speed_10MHz | GPIO_CNF_OUT_PP);
    funPinMode(LED_YELLOW, GPIO_Speed_10MHz | GPIO_CNF_OUT_PP);
    funPinMode(LED_ORANGE, GPIO_Speed_10MHz | GPIO_CNF_OUT_PP);

    // Start with only orange LED on (silence indicator)
    funDigitalWrite(LED_ORANGE, LED_ON);
    funDigitalWrite(LED_YELLOW, LED_OFF);
    funDigitalWrite(LED_BLUE, LED_OFF);
    funDigitalWrite(LED_PURPLE, LED_OFF);

    printf("Starting continuous reading...\n");
    printf("ADC | mV   | Dev | Status\n");
    printf("----|------|-----|-------\n");

    // Variables for tracking audio levels
    uint16_t baseline = 512;  // Expected DC bias point
    uint16_t max_deviation = 0;
    uint32_t sample_count = 0;

    while (1)
    {
        // Read current ADC value (DMA updates this automatically)
        uint16_t adc_val = adc_buffer[0];

        // Calculate deviation from baseline (DC bias)
        int16_t deviation = (int16_t)adc_val - (int16_t)baseline;
        if (deviation < 0) deviation = -deviation;  // Absolute value

        // Track maximum deviation (audio amplitude)
        if (deviation > max_deviation) {
            max_deviation = deviation;
        }

        sample_count++;

        // Every 100 samples (~100ms)
        if (sample_count >= 100) {
            // Calculate voltage in millivolts (integer math)
            uint16_t millivolts = (adc_val * 3300) / 1024;

            // Update baseline (slow moving average toward current reading)
            baseline = (baseline * 15 + adc_val) / 16;

            // Determine status character
            char status;
            if (max_deviation > 100) {
                status = 'L';  // LOUD
            }
            else if (max_deviation > 50) {
                status = 'S';  // Sound
            }
            else if (max_deviation > 20) {
                status = 'Q';  // Quiet
            }
            else {
                status = '-';  // Silence
            }

            // Print reading (integers only, no floats or strings)
            printf("%3d | %4d | %3d | %c\n",
                adc_val, millivolts, max_deviation, status);

            // Update LED indicator
            update_leds(max_deviation);

            // Reset for next measurement period
            max_deviation = 0;
            sample_count = 0;
        }

        // Small delay to not overwhelm the loop
        Delay_Ms(1);
    }
}
